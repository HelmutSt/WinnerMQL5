SELECT   exitReason, count(*) cnt, 
AVG(Profit) as avgProfit,
AVG(MAE) as avgMae,
sqrt(AVG(MAE * MAE) - AVG(MAE) * AVG(MAE)) AS stDevMae,

JOIN patternDynamic ON patternDynamic.eventId = events.eventId
JOIN trades ON trades.eventId = events.eventId

  AND events.patternTF = 'M3'
  AND events.eventReason = 'IS_TOUCHED'
  AND patternDynamic.causedOppositePatternBreak = 1

SELECT
    pc.type                                AS pattern_type,
    pd.status                              AS pattern_status,
    pd.touches,
    pd.isTrendBreak,
    pd.isStartOfMove,
    pd.sequenceSinceStartOfMove,
    pd.causedOppositePatternBreak,
    e.eventId,
m.sessionPhase,
m.session,
    r.slotName,
    r.isOriginPattern,
    t.eventId,
    t.barsWaitedForFill,
    t.barsHeldUntilExit,
    t.exitReason,
    t.profit,
    t.mae,
    t.mfe
FROM trades t
JOIN events e
    ON t.eventId = e.eventId
JOIN market m
    ON t.eventId = m.eventId
JOIN relations r
    ON t.eventId = r.eventId
LEFT JOIN patternDynamic pd
    ON r.patternId = pd.patternId and r.eventId = pd.eventId
LEFT JOIN patternCore pc
    ON e.patternId = pc.patternId
WHERE r.isOriginPattern < 3   
ORDER BY t.eventId
limit 10;

SELECT
m.session,
count(*),
sum(t.profit)
FROM trades t
JOIN events e
    ON t.eventId = e.eventId
JOIN market m
    ON t.eventId = m.eventId
JOIN relations r
    ON t.eventId = r.eventId
LEFT JOIN patternDynamic pd
    ON r.patternId = pd.patternId and r.eventId = pd.eventId
LEFT JOIN patternCore pc
    ON e.patternId = pc.patternId
WHERE r.isOriginPattern < 3   
ORDER BY m.session;
